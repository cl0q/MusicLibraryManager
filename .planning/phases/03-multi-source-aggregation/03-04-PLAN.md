```rust
use unicode_normalization::UnicodeNormalization;
use regex::Regex;

/// Normalize track title or album name for comparison
///
/// Handles:
/// - Unicode normalization (NFC: canonical composition)
/// - Lowercase conversion
/// - Punctuation removal
/// - Whitespace collapsing
///
/// Example: "Café!" -> "cafe"
pub fn normalize(text: &str) -> String {
    // Unicode normalization (NFC: compose diacritics)
    let nfc: String = text.nfc().collect();

    // Lowercase
    let lower = nfc.to_lowercase();

    // Remove punctuation, keep alphanumeric and whitespace
    let cleaned: String = lower
        .chars()
        .filter(|c| c.is_alphanumeric() || c.is_whitespace())
        .collect();

    // Collapse multiple spaces to single space
    cleaned.split_whitespace().collect::<Vec<_>>().join(" ").trim().to_string()
}

/// Normalize artist name with featuring notation standardization
///
/// Handles:
/// - All normalization from normalize()
/// - Featuring variations: "feat.", "ft.", "featuring", "& Artist" -> "[feat]"
/// - Removes parentheses around featuring notation
///
/// Example: "Artist (feat. Someone)" -> "artist [feat] someone"
pub fn normalize_artist(artist: &str) -> String {
    let normalized = normalize(artist);

    // Standardize featuring notation
    // Match variations: feat., ft., featuring, & (with word boundaries)
    let re_feat = Regex::new(r"\b(feat\.?|ft\.?|featuring|with)\b").unwrap();
    let with_marker = re_feat.replace_all(&normalized, "[feat]").to_string();

    // Replace "&" between words with [feat] marker
    let re_ampersand = Regex::new(r"\s+&\s+").unwrap();
    let standardized = re_ampersand.replace_all(&with_marker, " [feat] ").to_string();

    // Remove parentheses and brackets (now redundant after featuring standardization)
    let re_parens = Regex::new(r"[(){}\[\]]").unwrap();
    let cleaned = re_parens.replace_all(&standardized, "").to_string();

    // Final whitespace collapse
    cleaned.split_whitespace().collect::<Vec<_>>().join(" ").trim().to_string()
}
```

**matcher.rs** - Fuzzy duplicate detection (from RESEARCH.md Pattern 3):

```rust
use strsim::jaro_winkler;
use crate::dedup::normalize::{normalize, normalize_artist};
use crate::database::connection::Database;
use regex::Regex;

/// Calculate similarity score between two tracks
///
/// Returns value 0.0-1.0 where:
/// - 1.0 = exact match
/// - 0.95+ = very likely duplicate
/// - 0.85-0.95 = likely duplicate (check manually)
/// - < 0.85 = different tracks
///
/// Weighting: title 70%, artist 30%
pub fn calculate_similarity(
    title1: &str,
    artist1: &str,
    title2: &str,
    artist2: &str,
) -> f64 {
    let norm_title1 = normalize(title1);
    let norm_title2 = normalize(title2);
    let norm_artist1 = normalize_artist(artist1);
    let norm_artist2 = normalize_artist(artist2);

    // Empty strings are not equal (missing metadata)
    if norm_title1.is_empty() || norm_title2.is_empty() {
        return 0.0;
    }
    if norm_artist1.is_empty() || norm_artist2.is_empty() {
        return 0.0;
    }

    let title_sim = jaro_winkler(&norm_title1, &norm_title2);
    let artist_sim = jaro_winkler(&norm_artist1, &norm_artist2);

    // Weighted average: title more important than artist
    (title_sim * 0.7) + (artist_sim * 0.3)
}

/// Find potential duplicates in database for a given track
///
/// Returns list of (track_id, similarity_score) sorted by score descending
///
/// Threshold: 0.85 (from STATE.md learnings, slightly relaxed from Python's 0.80)
pub async fn find_duplicates(
    db: &Database,
    track_title: &str,
    artist: &str,
    threshold: f64,
) -> Result<Vec<(i64, f64)>, Box<dyn std::error::Error>> {
    // Get all tracks from database (Phase 1 search module has this)
    // For large libraries, use database LIKE for candidates first (two-pass search pattern)

    let candidates = db.get_all_tracks()?;

    let mut matches: Vec<(i64, f64)> = candidates
        .iter()
        .filter_map(|existing| {
            let score = calculate_similarity(
                track_title,
                artist,
                &existing.title,
                &existing.artist,
            );
            if score >= threshold {
                Some((existing.id, score))
            } else {
                None
            }
        })
        .collect();

    // Sort by score descending (highest similarity first)
    matches.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));

    Ok(matches)
}

/// Check if a track is a variant of another track (remix, edit, live, acoustic)
///
/// Returns true if title contains variant keywords but artist matches closely
///
/// Used for variant_of field in database (from CONTEXT.md decision)
pub fn is_variant(title1: &str, title2: &str, artist1: &str, artist2: &str) -> bool {
    let norm_title1 = normalize(title1);
    let norm_title2 = normalize(title2);
    let norm_artist1 = normalize_artist(artist1);
    let norm_artist2 = normalize_artist(artist2);

    // Artist must be similar (0.90+ threshold)
    let artist_sim = jaro_winkler(&norm_artist1, &norm_artist2);
    if artist_sim < 0.90 {
        return false;
    }

    // Check for variant keywords
    let variant_keywords = vec![
        "remix", "edit", "mix", "version", "live", "acoustic",
        "radio", "extended", "instrumental", "vocal", "dub",
        "vip", "bootleg", "rework", "remaster",
    ];

    let re_variants = Regex::new(&format!(r"\b({})\b", variant_keywords.join("|"))).unwrap();

    // One title has variant keyword, other doesn't
    let has_variant1 = re_variants.is_match(&norm_title1);
    let has_variant2 = re_variants.is_match(&norm_title2);

    if has_variant1 != has_variant2 {
        // Check if base title is similar (remove variant keywords and compare)
        let base1 = re_variants.replace_all(&norm_title1, "").trim().to_string();
        let base2 = re_variants.replace_all(&norm_title2, "").trim().to_string();

        let base_sim = jaro_winkler(&base1, &base2);
        return base_sim >= 0.85;  // High similarity on base title
    }

    false
}
```
