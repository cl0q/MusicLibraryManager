use tauri::State;
use crate::sources::{spotify::SpotifyClient, soundcloud::SoundCloudClient};
use crate::database::connection::Database;
use crate::dedup::matcher::find_duplicates;
use oauth2::{PkceCodeVerifier, CsrfToken};
use std::sync::Mutex;

/// State for OAuth PKCE verifiers (temporary storage during auth flow)
pub struct OAuthState {
    pub spotify_verifier: Mutex<Option<(PkceCodeVerifier, CsrfToken)>>,
    pub soundcloud_verifier: Mutex<Option<(PkceCodeVerifier, CsrfToken)>>,
}

impl Default for OAuthState {
    fn default() -> Self {
        Self {
            spotify_verifier: Mutex::new(None),
            soundcloud_verifier: Mutex::new(None),
        }
    }
}

// ============================================================================
// Spotify Commands
// ============================================================================

#[tauri::command]
pub fn spotify_auth_url(oauth_state: State<OAuthState>) -> Result<String, String> {
    let (url, csrf_token, pkce_verifier) = SpotifyClient::authorization_url();

    // Store verifier temporarily for code exchange
    let mut verifier = oauth_state.spotify_verifier.lock()
        .map_err(|e| format!("Failed to lock state: {}", e))?;
    *verifier = Some((pkce_verifier, csrf_token));

    Ok(url)
}

#[tauri::command]
pub async fn spotify_exchange_code(
    code: String,
    user_id: String,
    oauth_state: State<'_, OAuthState>,
) -> Result<(), String> {
    let verifier = {
        let mut state = oauth_state.spotify_verifier.lock()
            .map_err(|e| format!("Failed to lock state: {}", e))?;
        state.take()
            .ok_or("No PKCE verifier found. Call spotify_auth_url first.")?
    };

    SpotifyClient::exchange_code(&code, verifier.0, &user_id)
        .await
        .map_err(|e| format!("Token exchange failed: {}", e))?;

    Ok(())
}

#[tauri::command]
pub async fn sync_spotify(
    user_id: String,
    db: State<'_, Database>,
) -> Result<usize, String> {
    let client_id = std::env::var("SPOTIFY_CLIENT_ID")
        .map_err(|_| "SPOTIFY_CLIENT_ID not set".to_string())?;
    let client_secret = std::env::var("SPOTIFY_CLIENT_SECRET")
        .map_err(|_| "SPOTIFY_CLIENT_SECRET not set".to_string())?;

    let mut client = SpotifyClient::new(&client_id, &client_secret);

    // Call sync_liked_songs which updates last_sync_timestamps table
    let added_count = client.sync_liked_songs(&user_id, &db)
        .await
        .map_err(|e| format!("Sync failed: {}", e))?;

    Ok(added_count)
}

use crate::database::connection::Database;
use crate::sources::{spotify::SpotifyClient, soundcloud::SoundCloudClient};
use tauri::AppHandle;

/// Run startup tasks when app opens
///
/// Implements locked decision from CONTEXT.md: "Auto-sync on app open"
///
/// Tasks:
/// - Background sync of Spotify library (if authenticated)
/// - Background sync of SoundCloud library (if authenticated)
/// - Silent failure if user hasn't authenticated yet
pub async fn run_startup_tasks(db: &Database, user_id: &str) {
    // Attempt Spotify sync
    if let Ok(client_id) = std::env::var("SPOTIFY_CLIENT_ID") {
        if let Ok(client_secret) = std::env::var("SPOTIFY_CLIENT_SECRET") {
            let mut spotify_client = SpotifyClient::new(&client_id, &client_secret);

            // Silent sync - don't block startup if it fails
            match spotify_client.sync_liked_songs(user_id, db).await {
                Ok(count) => {
                    if count > 0 {
                        log::info!("Startup sync: Added {} new Spotify tracks", count);
                    }
                }
                Err(e) => {
                    log::warn!("Startup Spotify sync failed (user may not be authenticated): {}", e);
                }
            }
        }
    }

    // Attempt SoundCloud sync
    if let Ok(client_id) = std::env::var("SOUNDCLOUD_CLIENT_ID") {
        if let Ok(client_secret) = std::env::var("SOUNDCLOUD_CLIENT_SECRET") {
            let mut soundcloud_client = SoundCloudClient::new(&client_id, &client_secret);

            // Silent sync - don't block startup if it fails
            match soundcloud_client.sync_likes(user_id, db).await {
                Ok(count) => {
                    if count > 0 {
                        log::info!("Startup sync: Added {} new SoundCloud tracks", count);
                    }
                }
                Err(e) => {
                    log::warn!("Startup SoundCloud sync failed (user may not be authenticated): {}", e);
                }
            }
        }
    }
}
