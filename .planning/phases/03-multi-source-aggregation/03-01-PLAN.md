```sql
CREATE TABLE IF NOT EXISTS sources (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,  -- 'spotify' | 'soundcloud'
    user_id TEXT NOT NULL,       -- User identifier for this source
    enabled INTEGER DEFAULT 1,   -- 1 = active, 0 = disabled
    UNIQUE(name, user_id)
);
```

2. **track_sources table** - Many-to-many relationship between tracks and sources:
```sql
CREATE TABLE IF NOT EXISTS track_sources (
    track_id INTEGER NOT NULL,
    source_id INTEGER NOT NULL,  -- IMPORTANT: Source priority for downloads (CONTEXT.md)
    external_id TEXT NOT NULL,   -- Source's ID for this track (Spotify URI, SoundCloud ID)
    added_at TEXT NOT NULL,      -- When track was added to source (ISO 8601)
    PRIMARY KEY (track_id, source_id),
    FOREIGN KEY (track_id) REFERENCES tracks(id) ON DELETE CASCADE,
    FOREIGN KEY (source_id) REFERENCES sources(id) ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS idx_external_id ON track_sources(external_id);
```

3. **last_sync_timestamps table** - Track last sync time per source:
```sql
CREATE TABLE IF NOT EXISTS last_sync_timestamps (
    user_id TEXT NOT NULL,
    source TEXT NOT NULL,        -- 'spotify' | 'soundcloud'
    timestamp TEXT NOT NULL,     -- ISO 8601 datetime
    PRIMARY KEY (user_id, source)
);
```

**token_storage.rs** - Keychain integration (from RESEARCH.md Pattern 1):
```rust
use keyring::Entry;
use crate::database::connection::Result;

const SERVICE_NAME: &str = "com.musiclibrarymanager";

pub fn store_refresh_token(source: &str, user_id: &str, token: &str) -> Result<()> {
    let key = format!("{}_{}_refresh", source, user_id);
    let entry = Entry::new(SERVICE_NAME, &key)?;
    entry.set_password(token)?;
    Ok(())
}

pub fn get_refresh_token(source: &str, user_id: &str) -> Result<String> {
    let key = format!("{}_{}_refresh", source, user_id);
    let entry = Entry::new(SERVICE_NAME, &key)?;
    entry.get_password().map_err(|e| e.into())
}

pub fn delete_token(source: &str, user_id: &str) -> Result<()> {
    let key = format!("{}_{}_refresh", source, user_id);
    let entry = Entry::new(SERVICE_NAME, &key)?;
    entry.delete_credential()?;
    Ok(())
}
```

**token_refresh.rs** - Token lifecycle management:
```rust
use oauth2::{
    basic::BasicClient, AccessToken, AuthUrl, ClientId, ClientSecret,
    RefreshToken, TokenUrl, RedirectUrl,
};
use chrono::{DateTime, Utc, Duration};
use crate::auth::token_storage;
use crate::database::connection::Result;

pub struct TokenManager {
    client: BasicClient,
    source: String,
}

impl TokenManager {
    pub fn new(source: &str, client_id: &str, client_secret: &str, auth_url: &str, token_url: &str) -> Self {
        let client = BasicClient::new(
            ClientId::new(client_id.to_string()),
            Some(ClientSecret::new(client_secret.to_string())),
            AuthUrl::new(auth_url.to_string()).unwrap(),
            Some(TokenUrl::new(token_url.to_string()).unwrap()),
        )
        .set_redirect_uri(
            RedirectUrl::new("http://localhost:8080/callback".to_string()).unwrap(),
        );

        Self {
            client,
            source: source.to_string(),
        }
    }

    /// Ensure valid access token, refreshing if needed
    /// Returns new access token (refreshed if expired within 5 minutes)
    pub async fn ensure_valid_token(&self, user_id: &str, expiry: DateTime<Utc>) -> Result<AccessToken> {
        let now = Utc::now();

        // Refresh proactively if expiring within 5 minutes
        if now >= expiry - Duration::minutes(5) {
            self.refresh_access_token(user_id).await
        } else {
            // Token still valid, no need to refresh
            // Caller should have cached access token
            Err("Token refresh needed".into())
        }
    }

    async fn refresh_access_token(&self, user_id: &str) -> Result<AccessToken> {
        let refresh_token_str = token_storage::get_refresh_token(&self.source, user_id)?;
        let refresh_token = RefreshToken::new(refresh_token_str);

        let token_result = self.client
            .exchange_refresh_token(&refresh_token)
            .request_async(oauth2::reqwest::async_http_client)
            .await
            .map_err(|e| format!("Token refresh failed: {:?}", e))?;

        // Store new refresh token if provided (Spotify may rotate)
        if let Some(new_refresh) = token_result.refresh_token() {
            token_storage::store_refresh_token(&self.source, user_id, new_refresh.secret())?;
        }

        Ok(token_result.access_token().clone())
    }
}
```
