**soundcloud.rs** - SoundCloud API client (similar structure to Spotify, with OAuth 2.1 PKCE requirement):

IMPORTANT: SoundCloud requires PKCE (OAuth 2.1 compliance, deadline passed Oct 1, 2024). Always use PkceCodeChallenge::new_random_sha256().

1. **OAuth 2.1 Authorization Flow** with mandatory PKCE:
```rust
use oauth2::{
    basic::BasicClient, AuthUrl, ClientId, ClientSecret, CsrfToken,
    PkceCodeChallenge, PkceCodeVerifier, RedirectUrl, TokenUrl,
    AuthorizationCode, Scope,
};
use reqwest::Client;
use serde::{Deserialize, Serialize};
use chrono::{DateTime, Utc, Duration};
use crate::auth::{TokenManager, store_refresh_token};
use crate::database::connection::Database;

const AUTH_URL: &str = "https://soundcloud.com/connect";
const TOKEN_URL: &str = "https://api.soundcloud.com/oauth2/token";
const API_BASE: &str = "https://api.soundcloud.com";

pub struct SoundCloudClient {
    http: Client,
    token_manager: TokenManager,
    access_token: String,
    token_expiry: DateTime<Utc>,
}

impl SoundCloudClient {
    pub fn new(client_id: &str, client_secret: &str) -> Self {
        let token_manager = TokenManager::new("soundcloud", client_id, client_secret, AUTH_URL, TOKEN_URL);

        Self {
            http: Client::new(),
            token_manager,
            access_token: String::new(),
            token_expiry: Utc::now(),
        }
    }

    /// Generate OAuth authorization URL with PKCE (mandatory for OAuth 2.1)
    pub fn authorization_url() -> (String, CsrfToken, PkceCodeVerifier) {
        let client = BasicClient::new(
            ClientId::new(std::env::var("SOUNDCLOUD_CLIENT_ID").unwrap()),
            Some(ClientSecret::new(std::env::var("SOUNDCLOUD_CLIENT_SECRET").unwrap())),
            AuthUrl::new(AUTH_URL.to_string()).unwrap(),
            Some(TokenUrl::new(TOKEN_URL.to_string()).unwrap()),
        )
        .set_redirect_uri(RedirectUrl::new("http://localhost:8080/callback".to_string()).unwrap());

        let (pkce_challenge, pkce_verifier) = PkceCodeChallenge::new_random_sha256();

        let (auth_url, csrf_token) = client
            .authorize_url(CsrfToken::new_random)
            .add_scope(Scope::new("non-expiring".to_string()))  // SoundCloud scope for persistent access
            .set_pkce_challenge(pkce_challenge)
            .url();

        (auth_url.to_string(), csrf_token, pkce_verifier)
    }

    /// Exchange authorization code for tokens
    pub async fn exchange_code(code: &str, pkce_verifier: PkceCodeVerifier, user_id: &str) -> Result<(), Box<dyn std::error::Error>> {
        let client_id = std::env::var("SOUNDCLOUD_CLIENT_ID")?;
        let client_secret = std::env::var("SOUNDCLOUD_CLIENT_SECRET")?;

        let client = BasicClient::new(
            ClientId::new(client_id),
            Some(ClientSecret::new(client_secret)),
            AuthUrl::new(AUTH_URL.to_string())?,
            Some(TokenUrl::new(TOKEN_URL.to_string())?),
        )
        .set_redirect_uri(RedirectUrl::new("http://localhost:8080/callback".to_string())?);

        let token_result = client
            .exchange_code(AuthorizationCode::new(code.to_string()))
            .set_pkce_verifier(pkce_verifier)
            .request_async(oauth2::reqwest::async_http_client)
            .await?;

        // Store refresh token in keychain
        if let Some(refresh_token) = token_result.refresh_token() {
            store_refresh_token("soundcloud", user_id, refresh_token.secret())?;
        }

        Ok(())
    }

    /// Ensure access token is valid, refresh if needed
    async fn ensure_token(&mut self, user_id: &str) -> Result<(), Box<dyn std::error::Error>> {
        if Utc::now() >= self.token_expiry - Duration::minutes(5) {
            let new_token = self.token_manager.ensure_valid_token(user_id, self.token_expiry).await?;
            self.access_token = new_token.secret().clone();
            self.token_expiry = Utc::now() + Duration::hours(1);
        }
        Ok(())
    }
}
```
