use std::path::PathBuf;
use tokio::process::Command;
use serde::{Deserialize, Serialize};
use crate::database::connection::Result;
use crate::auth::token_storage;

#[derive(Debug, Serialize, Deserialize)]
pub struct SoundCloudDownloadRequest {
    pub track_url: String,        // SoundCloud permalink URL
    pub user_id: String,           // For keychain access token retrieval
    pub output_dir: PathBuf,
    pub track_id: String,          // For filename generation
}

/// Download track from SoundCloud using scdl CLI
///
/// Requires:
/// - scdl installed (`pip install scdl`)
/// - SoundCloud Go+ auth token stored in keychain
///
/// Returns path to downloaded file (248kbps AAC format)
pub async fn download_soundcloud_track(request: &SoundCloudDownloadRequest) -> Result<PathBuf> {
    // Retrieve SoundCloud access token from keychain
    let access_token = token_storage::get_refresh_token("soundcloud", &request.user_id)
        .map_err(|e| format!("SoundCloud auth token not found: {}. Run OAuth flow first.", e))?;

    // Ensure output directory exists
    tokio::fs::create_dir_all(&request.output_dir).await?;

    // Build scdl command
    // scdl -l <track_url> --auth-token <token> --path <output_dir> --onlymp3 --original-art
    let output = Command::new("scdl")
        .arg("-l")
        .arg(&request.track_url)
        .arg("--auth-token")
        .arg(&access_token)
        .arg("--path")
        .arg(&request.output_dir)
        .arg("--onlymp3")                  // Force AAC/MP3 output (not FLAC)
        .arg("--original-art")             // Download cover art
        .arg("--name-format")
        .arg("{title}")                    // Simple filename (we'll rename)
        .output()
        .await
        .map_err(|e| format!("Failed to execute scdl: {}. Is scdl installed?", e))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("scdl failed: {}", stderr).into());
    }

    // scdl downloads to output_dir with pattern "{title}.mp3"
    // Find the downloaded file (most recent .mp3 in output_dir)
    let downloaded_file = find_most_recent_audio_file(&request.output_dir).await?;

    Ok(downloaded_file)
}

/// Find most recently modified audio file in directory
///
/// Helper for locating scdl output when exact filename is unknown
async fn find_most_recent_audio_file(dir: &PathBuf) -> Result<PathBuf> {
    let mut entries = tokio::fs::read_dir(dir).await?;
    let mut candidates = Vec::new();

    while let Some(entry) = entries.next_entry().await? {
        let path = entry.path();
        if let Some(ext) = path.extension() {
            if ext == "mp3" || ext == "m4a" {
                let metadata = entry.metadata().await?;
                if let Ok(modified) = metadata.modified() {
                    candidates.push((path, modified));
                }
            }
        }
    }

    candidates.sort_by(|a, b| b.1.cmp(&a.1));  // Sort by modified time descending

    candidates
        .first()
        .map(|(path, _)| path.clone())
        .ok_or_else(|| "No audio file found in download directory".into())
}

## Dependencies

### Python Tools (for SoundCloud downloads)

SoundCloud direct downloads require `scdl` CLI tool:

```bash
pip install scdl
```

**SoundCloud Go+ Subscription Required:**
- High-quality (248kbps AAC) downloads require SoundCloud Go+ subscription
- Free accounts are limited to 128kbps streams
- Authenticate via OAuth flow in app to store access token

### Download Source Priority

The app attempts downloads in this order based on source:

**Spotify tracks:**
1. DAB (FLAC) - lossless quality
2. YouTube (fallback) - 320kbps MP3

**SoundCloud tracks:**
1. SoundCloud direct (scdl) - 248kbps AAC with Go+
2. DAB (FLAC) - if track is available
3. YouTube (fallback) - 320kbps MP3

Source information is tracked in `track_sources` table to guide priority.
