- album (TEXT, stored=True, phrase=False)
- title (TEXT, stored=True, phrase=False)
- genre (TEXT, stored=True)
- year (NUMERIC, stored=True)
- comments (TEXT, stored=True)

Functions to implement:

1. create_index() -> Index:
   - Create Whoosh index in config.WHOOSH_INDEX_DIR
   - Define schema with all searchable fields
   - Return index object

2. index_file(file_id: int) -> bool:
   - Fetch file metadata from database
   - Add document to Whoosh index
   - Update search_index table with indexed_at timestamp
   - Return True on success

3. index_library() -> int:
   - Fetch all files from database
   - Index each file's metadata
   - Update search_index table for tracking
   - Return count of indexed files

4. reindex_library() -> int:
   - Clear existing index
   - Recreate and index all files
   - Return count of indexed files

Use config.WHOOSH_INDEX_DIR for index location.

Example structure:
```python
from whoosh.fields import Schema, TEXT, NUMERIC, STORED
from whoosh.index import create_in, open_dir, exists_in
from pathlib import Path
from src.library.database import get_all_files, get_file_by_id
from src.core.config import WHOOSH_INDEX_DIR

def get_schema() -> Schema:
    """Define Whoosh schema for music library search."""
    return Schema(
        file_id=NUMERIC(stored=True, unique=True),
        artist=TEXT(stored=True, phrase=False),
        album_artist=TEXT(stored=True, phrase=False),
        album=TEXT(stored=True, phrase=False),
        title=TEXT(stored=True, phrase=False),
        genre=TEXT(stored=True),
        year=NUMERIC(stored=True),
        comments=TEXT(stored=True)
    )

def create_index():
    """Create new Whoosh index."""
    WHOOSH_INDEX_DIR.mkdir(parents=True, exist_ok=True)
    schema = get_schema()
    return create_in(str(WHOOSH_INDEX_DIR), schema)

def get_index():
    """Get existing index or create new one."""
    if exists_in(str(WHOOSH_INDEX_DIR)):
        return open_dir(str(WHOOSH_INDEX_DIR))
    else:
        return create_index()

def index_file(file_id: int) -> bool:
    """Index single file metadata."""
    file_record = get_file_by_id(file_id)
    if not file_record:
        return False

    metadata = file_record['metadata']
    ix = get_index()
    writer = ix.writer()

    try:
        writer.add_document(
            file_id=file_id,
            artist=metadata.get('artist', ''),
            album_artist=metadata.get('album_artist', ''),
            album=metadata.get('album', ''),
            title=metadata.get('title', ''),
            genre=metadata.get('genre', ''),
            year=metadata.get('year', 0),
            comments=metadata.get('comments', '')
        )
        writer.commit()
        return True
    except Exception as e:
        writer.cancel()
        raise

def index_library() -> int:
    """Index all files in library."""
    files = get_all_files()
    ix = get_index()
    writer = ix.writer()

    indexed_count = 0
    try:
        for file_record in files:
            metadata = file_record['metadata']
            writer.add_document(
                file_id=file_record['id'],
                artist=metadata.get('artist', ''),
                album_artist=metadata.get('album_artist', ''),
                album=metadata.get('album', ''),
                title=metadata.get('title', ''),
                genre=metadata.get('genre', ''),
                year=metadata.get('year', 0),
                comments=metadata.get('comments', '')
            )
            indexed_count += 1
        writer.commit()
        return indexed_count
    except Exception as e:
        writer.cancel()
        raise

def reindex_library() -> int:
    """Clear and rebuild entire index."""
    # Clear existing index
    if WHOOSH_INDEX_DIR.exists():
        import shutil
        shutil.rmtree(WHOOSH_INDEX_DIR)

    # Recreate
    create_index()
    return index_library()
```

Use dataclasses for clean structure:
```python
from dataclasses import dataclass
from typing import List

@dataclass
class SearchResult:
    """Single search result."""
    file_id: int
    artist: str
    album_artist: str
    album: str
    title: str
    genre: str
    year: int
    score: float
    original_path: str
    organized_path: str

    def __str__(self) -> str:
        return f"{self.artist} - {self.title} (from {self.album}) [score: {self.score:.2f}]"

@dataclass
class SearchResponse:
    """Search operation result."""
    query: str
    total_results: int
    results: List[SearchResult]
    search_time_ms: float

    def __str__(self) -> str:
        return f"Found {self.total_results} results for '{self.query}' in {self.search_time_ms:.2f}ms"
```

Create search engine combining Whoosh full-text search with RapidFuzz fuzzy matching.

Follow RESEARCH.md Pattern 4 and Code Example (Fuzzy Search with RapidFuzz).

Main function: search_library(query: str, limit: int = 50) -> SearchResponse

Search strategy:
1. Query Whoosh index for initial candidates (fast full-text search)
2. Apply RapidFuzz fuzzy matching to candidates with token_sort_ratio
3. Filter by threshold (config.SEARCH_FUZZY_THRESHOLD, default 80)
4. Sort by score descending
5. Limit results
6. Return SearchResponse with timing

Use RapidFuzz token_sort_ratio (RESEARCH.md recommendation):
- Handles word reordering: "The Beatles" matches "Beatles, The"
- Better for metadata search than simple ratio

Example structure:
```python
from whoosh.qparser import MultifieldParser
from rapidfuzz import fuzz
from src.search.indexer import get_index
from src.search.models import SearchResult, SearchResponse
from src.library.database import get_file_by_id
from src.core.config import SEARCH_FUZZY_THRESHOLD
import time

def search_library(query: str, limit: int = 50) -> SearchResponse:
    """Search library with fuzzy matching."""
    start_time = time.time()

    ix = get_index()
    searcher = ix.searcher()

    # Parse query for multiple fields
    parser = MultifieldParser(["artist", "album_artist", "album", "title"], ix.schema)
    whoosh_query = parser.parse(query)

    # Get initial candidates from Whoosh
    whoosh_results = searcher.search(whoosh_query, limit=limit * 2)  # Get more for fuzzy filtering

    # Apply fuzzy matching
    fuzzy_results = []
    for hit in whoosh_results:
        # Combine searchable fields into single string
        searchable_text = f"{hit.get('artist', '')} {hit.get('album_artist', '')} {hit.get('album', '')} {hit.get('title', '')}".lower()

        # Calculate fuzzy score
        score = fuzz.token_sort_ratio(query.lower(), searchable_text)

        if score >= SEARCH_FUZZY_THRESHOLD:
            # Fetch full file record for paths
            file_record = get_file_by_id(hit['file_id'])
            if file_record:
                result = SearchResult(
                    file_id=hit['file_id'],
                    artist=hit.get('artist', ''),
                    album_artist=hit.get('album_artist', ''),
                    album=hit.get('album', ''),
                    title=hit.get('title', ''),
                    genre=hit.get('genre', ''),
                    year=hit.get('year', 0),
                    score=score,
                    original_path=file_record['original_path'],
                    organized_path=file_record['organized_path']
                )
                fuzzy_results.append(result)

    # Sort by score descending
    fuzzy_results.sort(key=lambda r: r.score, reverse=True)

    # Limit results
    fuzzy_results = fuzzy_results[:limit]

    elapsed_ms = (time.time() - start_time) * 1000

    return SearchResponse(
        query=query,
        total_results=len(fuzzy_results),
        results=fuzzy_results,
        search_time_ms=elapsed_ms
    )

def search_by_artist(artist: str, limit: int = 50) -> SearchResponse:
    """Search library by artist name."""
    return search_library(f"artist:{artist}", limit)

def search_by_album(album: str, limit: int = 50) -> SearchResponse:
    """Search library by album name."""
    return search_library(f"album:{album}", limit)

def search_by_title(title: str, limit: int = 50) -> SearchResponse:
    """Search library by track title."""
    return search_library(f"title:{title}", limit)
```
