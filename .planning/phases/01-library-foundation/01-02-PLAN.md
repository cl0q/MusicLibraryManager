```rust
use lofty::prelude::*;
use std::path::Path;
use crate::models::track::TrackMetadata;

pub fn extract_metadata(path: &Path) -> Result&lt;TrackMetadata, String&gt; {
    let format_str = path.extension()
        .and_then(|s| s.to_str())
        .unwrap_or("")
        .to_lowercase();

    let tagged_file = Probe::open(path)
        .map_err(|e| format!("Cannot open file: {}", e))?
        .read()
        .map_err(|e| format!("Cannot read file: {}", e))?;

    let tag = tagged_file.primary_tag()
        .or_else(|| tagged_file.first_tag())
        .ok_or("No metadata tags found")?;

    // Extract with fallbacks
    let artist = tag.artist()
        .and_then(|mut a| a.next())
        .map(|s| s.to_string());

    let album_artist = tag.album_artist()
        .and_then(|mut a| a.next())
        .map(|s| s.to_string())
        .or(artist.clone());

    let album = tag.album()
        .and_then(|mut a| a.next())
        .map(|s| s.to_string());

    let title = tag.title()
        .and_then(|mut t| t.next())
        .map(|s| s.to_string());

    // Normalize and apply fallbacks
    let artist_normalized = artist.map(|s| s.trim().to_lowercase()).unwrap_or_default();
    let album_artist_normalized = album_artist
        .map(|s| s.trim().to_lowercase())
        .or_else(|| if !artist_normalized.is_empty() { Some(artist_normalized.clone()) } else { None })
        .unwrap_or_else(|| "various artists".to_string());
    let album_normalized = album.map(|s| s.trim().to_lowercase())
        .unwrap_or_else(|| "unknown album".to_string());
    let title_normalized = title.map(|s| s.trim().to_lowercase())
        .unwrap_or_else(|| path.file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or("unknown")
            .to_string());

    // Extract audio properties
    let bitrate = tagged_file.properties().audio_bitrate();
    let duration = tagged_file.properties().duration().map(|d| d.as_secs() as u32);
    let genre = tag.genre().and_then(|mut g| g.next()).map(|s| s.to_string());
    let year = tag.year();

    Ok(TrackMetadata {
        artist: artist_normalized,
        album_artist: album_artist_normalized,
        album: album_normalized,
        title: title_normalized,
        genre,
        year,
        bitrate,
        duration,
        format: format_str,
        original_path: path.to_string_lossy().to_string(),
    })
}
```

```rust
use sanitize_filename::{sanitize_with_options, Options};

pub fn sanitize_filename(name: &str) -> String {
    if name.trim().is_empty() {
        return "unknown".to_string();
    }

    let options = Options {
        truncate: true,
        windows: true, // FAT32 compatibility for iPod
        replacement: "_"
    };

    let sanitized = sanitize_with_options(name, options);

    // Handle Windows reserved names (CON, PRN, AUX, etc.)
    if is_windows_reserved(&sanitized) {
        format!("_{}", sanitized)
    } else {
        sanitized
    }
}

fn is_windows_reserved(name: &str) -> bool {
    let upper = name.to_uppercase();
    matches!(
        upper.as_str(),
        "CON" | "PRN" | "AUX" | "NUL" | "COM1" | "COM2" | "COM3" | "COM4"
            | "COM5" | "COM6" | "COM7" | "COM8" | "COM9" | "LPT1" | "LPT2"
            | "LPT3" | "LPT4" | "LPT5" | "LPT6" | "LPT7" | "LPT8" | "LPT9"
    )
}

pub fn generate_organized_path(metadata: &crate::models::track::TrackMetadata) -> String {
    let artist = sanitize_filename(&metadata.album_artist);
    let album = sanitize_filename(&metadata.album);
    let title = sanitize_filename(&metadata.title);
    let ext = &metadata.format;

    // Format: Artist/Album/Track.ext
    format!("{}/{}/{}.{}", artist, album, title, ext)
}

// SoundCloud content goes to separate folder (LIB-06)
pub fn generate_soundcloud_path(metadata: &crate::models::track::TrackMetadata) -> String {
    let artist = sanitize_filename(&metadata.artist);
    let title = sanitize_filename(&metadata.title);
    let ext = &metadata.format;

    format!("SoundCloud/{}/{}.{}", artist, title, ext)
}
```
