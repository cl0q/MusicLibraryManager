Duplicate detection logic from CONTEXT.md and RESEARCH.md Common Pitfall 5:

**Exact metadata match:** artist + title + album must match exactly (case-insensitive)
**Quality hierarchy:**
1. Format: lossless (flac) > lossy (mp3, aac, m4a, ogg)
2. Bitrate: if both lossy, higher bitrate wins
3. If equal quality: keep first imported (earlier timestamp)

**Action:** Mark lower quality as duplicate, store reason ("lower_bitrate", "worse_format")
**Organization:** Create manifest or symlinks in _duplicates review folder per CONTEXT.md (not discarded)

Test cases:
1. Same track, different formats:
   - Input: MP3 320kbps, FLAC (same artist/title/album)
   - Output: MP3 marked as duplicate (reason: "worse_format")
   - Primary: FLAC

2. Same track, different bitrates (both lossy):
   - Input: MP3 128kbps, MP3 320kbps (same metadata)
   - Output: 128kbps marked as duplicate (reason: "lower_bitrate")
   - Primary: 320kbps

3. Exact same quality:
   - Input: MP3 320kbps imported at 2024-01-01, MP3 320kbps imported at 2024-01-02 (same metadata)
   - Output: Second import marked as duplicate (reason: "duplicate_of_earlier_import")
   - Primary: First import

4. Different titles (not duplicates):
   - Input: "Song (Live)", "Song" (different title metadata)
   - Output: NO duplicate detected (titles differ)

5. No duplicates:
   - Input: Single MP3
   - Output: No duplicates found

6. Multiple duplicates of same track:
   - Input: FLAC, MP3 320, MP3 128 (same metadata)
   - Output: MP3 320 and 128 both marked as duplicates of FLAC
   - Primary: FLAC

Edge case: Case-insensitive matching
   - Input: "The Beatles" vs "the beatles" (different case, same metadata otherwise)
   - Output: Duplicate detected (case-insensitive match)

```python
from pathlib import Path
from src.library.database import get_all_files, mark_as_duplicate, get_config

LOSSLESS_FORMATS = ['flac', 'wav', 'alac', 'ape']

def is_duplicate(meta1: dict, meta2: dict) -> bool:
    """Check if two tracks are duplicates (exact metadata match, case-insensitive)."""
    artist1 = meta1.get('artist', '').lower()
    artist2 = meta2.get('artist', '').lower()
    title1 = meta1.get('title', '').lower()
    title2 = meta2.get('title', '').lower()
    album1 = meta1.get('album', '').lower()
    album2 = meta2.get('album', '').lower()

    return artist1 == artist2 and title1 == title2 and album1 == album2

def compare_quality(meta1: dict, meta2: dict) -> str:
    """Compare quality of two tracks. Returns 'file1_better', 'file2_better', or 'equal'."""
    format1 = meta1.get('format', '').lower()
    format2 = meta2.get('format', '').lower()

    # Check format tier (lossless vs lossy)
    is_lossless1 = format1 in LOSSLESS_FORMATS
    is_lossless2 = format2 in LOSSLESS_FORMATS

    if is_lossless1 and not is_lossless2:
        return "file1_better"
    elif is_lossless2 and not is_lossless1:
        return "file2_better"

    # Both same tier, compare bitrate
    bitrate1 = meta1.get('bitrate', 0)
    bitrate2 = meta2.get('bitrate', 0)

    if bitrate1 > bitrate2:
        return "file1_better"
    elif bitrate2 > bitrate1:
        return "file2_better"

    return "equal"

def find_duplicates(file_id: int) -> list[tuple[int, str]]:
    """Find all duplicates of given file. Returns list of (duplicate_file_id, reason)."""
    all_files = get_all_files()
    target_file = next((f for f in all_files if f['id'] == file_id), None)

    if not target_file:
        return []

    duplicates = []

    for candidate in all_files:
        if candidate['id'] == file_id:
            continue

        if is_duplicate(target_file['metadata'], candidate['metadata']):
            quality = compare_quality(target_file['metadata'], candidate['metadata'])

            if quality == "file1_better":
                # Target is better, candidate is duplicate
                reason = _determine_reason(target_file['metadata'], candidate['metadata'])
                duplicates.append((candidate['id'], reason))
            elif quality == "file2_better":
                # Candidate is better, target is duplicate
                # Don't add to duplicates list (this function finds duplicates OF target)
                pass
            else:
                # Equal quality, use timestamp (imported_at)
                if target_file['imported_at'] < candidate['imported_at']:
                    duplicates.append((candidate['id'], "duplicate_of_earlier_import"))

    return duplicates

def _determine_reason(primary_meta: dict, duplicate_meta: dict) -> str:
    """Determine why duplicate is lower quality."""
    format1 = primary_meta.get('format', '').lower()
    format2 = duplicate_meta.get('format', '').lower()

    if format1 in LOSSLESS_FORMATS and format2 not in LOSSLESS_FORMATS:
        return "worse_format"

    bitrate1 = primary_meta.get('bitrate', 0)
    bitrate2 = duplicate_meta.get('bitrate', 0)

    if bitrate1 > bitrate2:
        return "lower_bitrate"

    return "duplicate"

def organize_duplicates_for_review():
    """Create manifest in _duplicates folder for user review."""
    # Get duplicates folder from config (Plan 01)
    config = get_config()
    duplicates_folder = Path(config.get('DUPLICATES_FOLDER', '_duplicates'))
    duplicates_folder.mkdir(exist_ok=True)

    # Get all marked duplicates from database
    from src.library.database import get_duplicates
    duplicates = get_duplicates(reviewed=False)

    if not duplicates:
        return

    # Create manifest file
    manifest_path = duplicates_folder / "duplicates_manifest.txt"
    with open(manifest_path, 'w') as f:
        f.write("# Duplicate Files Review\n")
        f.write("# Lower quality duplicates detected by metadata matching\n\n")

        for dup in duplicates:
            f.write(f"Primary: {dup['primary_file_path']}\n")
            f.write(f"Duplicate: {dup['duplicate_file_path']}\n")
            f.write(f"Reason: {dup['reason']}\n")
            f.write(f"---\n\n")

def mark_all_duplicates() -> int:
    """Scan entire library for duplicates, mark them, and organize for review. Returns count of duplicates marked."""
    all_files = get_all_files()
    marked_count = 0

    for file in all_files:
        duplicates = find_duplicates(file['id'])
        for duplicate_id, reason in duplicates:
            mark_as_duplicate(file['id'], duplicate_id, reason)
            marked_count += 1

    # Organize duplicates in review folder (CONTEXT.md requirement)
    if marked_count > 0:
        organize_duplicates_for_review()

    return marked_count
```
