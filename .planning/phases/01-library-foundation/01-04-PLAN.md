```rust
use tantivy::{
    schema::{Schema, STORED, TEXT},
    Index, IndexWriter, TantivyError, Document,
};
use std::path::Path;
use crate::models::track::Track;

pub struct SearchIndex {
    pub index: Index,
    pub artist_field: tantivy::schema::Field,
    pub album_field: tantivy::schema::Field,
    pub title_field: tantivy::schema::Field,
    pub id_field: tantivy::schema::Field,
}

pub fn create_index(index_path: &Path) -> Result&lt;SearchIndex, TantivyError&gt; {
    let mut schema_builder = Schema::builder();

    let artist_field = schema_builder.add_text_field("artist", TEXT | STORED);
    let album_field = schema_builder.add_text_field("album", TEXT | STORED);
    let title_field = schema_builder.add_text_field("title", TEXT | STORED);
    let id_field = schema_builder.add_u64_field("track_id", STORED);

    let schema = schema_builder.build();

    // Create index in specified directory (persistent storage)
    let index = Index::create_in_dir(index_path, schema)?;

    Ok(SearchIndex {
        index,
        artist_field,
        album_field,
        title_field,
        id_field,
    })
}

pub fn index_track(
    writer: &mut IndexWriter,
    track: &Track,
    search_index: &SearchIndex,
) -> Result&lt;(), TantivyError&gt; {
    let mut doc = Document::new();

    doc.add_text(search_index.artist_field, &track.metadata.artist);
    doc.add_text(search_index.album_field, &track.metadata.album);
    doc.add_text(search_index.title_field, &track.metadata.title);

    if let Some(id) = track.id {
        doc.add_u64(search_index.id_field, id as u64);
    }

    writer.add_document(doc)?;
    Ok(())
}

pub fn commit_index(writer: &mut IndexWriter) -> Result&lt;(), TantivyError&gt; {
    writer.commit()?;
    Ok(())
}
```

Implement pattern from RESEARCH.md Pattern 4 (Fuzzy Matching) + Code Example (Search with Fuzzy Scoring):
```rust
use rusqlite::Connection;
use strsim::{jaro_winkler, levenshtein};
use crate::database::Result as DbResult;
use crate::models::track::Track;

#[derive(Debug, serde::Serialize)]
pub struct SearchResult {
    pub track: Track,
    pub score: f64,
}

const FUZZY_THRESHOLD: f64 = 0.75;

pub fn search_tracks(
    conn: &Connection,
    query: &str,
) -> DbResult&lt;Vec&lt;SearchResult&gt;&gt; {
    // First pass: get all tracks from database (Tantivy integration in Phase 6)
    // For Phase 1, simple database query for candidates
    let mut stmt = conn.prepare(
        "SELECT id, artist, album_artist, album, title, genre, year, bitrate, duration, format, original_path, organized_path, is_duplicate
         FROM tracks
         WHERE artist LIKE ?1 OR album LIKE ?1 OR title LIKE ?1
         LIMIT 100"
    )?;

    let query_pattern = format!("%{}%", query);
    let candidates: Vec&lt;Track&gt; = stmt
        .query_map([&query_pattern], |row| {
            Ok(Track {
                id: Some(row.get(0)?),
                metadata: crate::models::track::TrackMetadata {
                    artist: row.get(1)?,
                    album_artist: row.get(2)?,
                    album: row.get(3)?,
                    title: row.get(4)?,
                    genre: row.get(5)?,
                    year: row.get(6)?,
                    bitrate: row.get(7)?,
                    duration: row.get(8)?,
                    format: row.get(9)?,
                    original_path: row.get(10)?,
                },
                organized_path: row.get(11)?,
                is_duplicate: row.get::<_, i32>(12)? != 0,
                date_added: None,
            })
        })?
        .filter_map(|r| r.ok())
        .collect();

    // Second pass: fuzzy score each candidate
    let query_lower = query.to_lowercase();
    let mut results: Vec&lt;SearchResult&gt; = candidates
        .into_iter()
        .map(|track| {
            let score = calculate_fuzzy_score(&query_lower, &track);
            SearchResult { track, score }
        })
        .filter(|result| result.score >= FUZZY_THRESHOLD)
        .collect();

    // Sort by score descending
    results.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(std::cmp::Ordering::Equal));

    Ok(results)
}

fn calculate_fuzzy_score(query: &str, track: &Track) -> f64 {
    // Score against each field (artist, album, title) and take best match
    let fields = [
        &track.metadata.artist,
        &track.metadata.album,
        &track.metadata.title,
    ];

    fields
        .iter()
        .map(|field| {
            let field_lower = field.to_lowercase();

            // Use both algorithms, take max (Python pattern replacement)
            let jaro = jaro_winkler(&query, &field_lower);
            let lev = normalize_levenshtein(query, &field_lower);

            jaro.max(lev)
        })
        .fold(0.0, f64::max)
}

fn normalize_levenshtein(s1: &str, s2: &str) -> f64 {
    let max_len = s1.len().max(s2.len()) as f64;
    if max_len == 0.0 {
        return 1.0;
    }
    let dist = levenshtein(s1, s2) as f64;
    1.0 - (dist / max_len)
}
```
