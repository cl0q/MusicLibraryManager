func togglePin(id: Int64) async {
    // D-10: Soft limit of 8 pinned playlists. Hard-block 9th attempt with a 3s hint.
    guard let target = playlists.first(where: { $0.id == id }) else { return }
    let willPin = target.isPinned == 0

    if willPin {
        let pinnedCount = playlists.filter { $0.isPinned == 1 }.count
        if pinnedCount >= 8 {
            pinLimitHintMessage = "Pinned limit reached"   // banner title; body lives in the View
            // Schedule auto-clear after 3 seconds (UI-SPEC line 241)
            Task { @MainActor [weak self] in
                try? await Task.sleep(for: .seconds(3))
                self?.pinLimitHintMessage = nil
            }
            return   // hard block — no repo call, no notification, no UI change beyond banner
        }
    }

    // Unchanged: existing pin flip + local state sync + sort + applyFilter
    do {
        try await playlistRepository.togglePin(id: id)
        if let idx = playlists.firstIndex(where: { $0.id == id }) {
            playlists[idx].isPinned = playlists[idx].isPinned == 1 ? 0 : 1
        }
        playlists.sort { a, b in
            if a.isPinned != b.isPinned { return a.isPinned > b.isPinned }
            return a.name.localizedCaseInsensitiveCompare(b.name) == .orderedAscending

```swift
private func handleDrop(providers: [NSItemProvider]) -> Bool {
    guard let provider = providers.first else {
        onCoverDropRejected()
        return false
    }

    // Prefer fileURL (Finder drop). Fall back to image (in-app drag).
    if provider.hasItemConformingToTypeIdentifier("public.file-url") {
        _ = provider.loadDataRepresentation(for: .fileURL) { data, _ in
            guard let data,
                  let url = URL(dataRepresentation: data, isAbsolute: true)
                       ?? URL(string: String(decoding: data, as: UTF8.self)) else {
                Task { @MainActor in onCoverDropRejected() }
                return
            }
            Task { await onCoverDropped(url) }
        }
        return true
    } else if provider.hasItemConformingToTypeIdentifier("public.image") {
        _ = provider.loadDataRepresentation(for: .image) { data, _ in
            guard let data else {
                Task { @MainActor in onCoverDropRejected() }
                return
            }
            // Write the in-app image to a temp file and forward as URL — same surface
            // as a Finder drop so PlaylistCoverService has a uniform input.
            let tmp = FileManager.default.temporaryDirectory
                .appendingPathComponent("inapp_cover_\(UUID().uuidString)")
            try? data.write(to: tmp)
            Task { await onCoverDropped(tmp) }
        }
        return true
    }
    // Non-image provider (e.g., text, folder)
    onCoverDropRejected()
    return false
}
```

The closure dispatches the path to the parent (`PlaylistsView`), which calls `container.playlistCoverService?.setCustomCover(playlistId:, sourceURL:)`. Rejected drops bubble up via `onCoverDropRejected` to the PlaylistsView banner.
