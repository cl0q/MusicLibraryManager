```swift
import SwiftUI

struct SyncProfileDetailView: View {
    let profile: SyncProfile
    @Environment(\.container) var container
    @State private var showPlaylistPicker = false
    @State private var showTrackPicker = false

    private var vm: SyncViewModel? { container.syncViewModel }
    private var syncService: SyncService? { container.syncService }

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 16) {
                // Header — keep existing header HStack from SyncView
                headerSection

                Divider().background(Color.mlmEdge)

                // Settings form (collapsed by default)
                SyncSettingsForm(profile: profile, vm: vm)

                Divider().background(Color.mlmEdgeSubtle)

                // Content: Add buttons + Playlists + Tracks sections
                contentHeader  // "Add Playlists…" and "Add Tracks…" buttons
                SyncContentSections(profile: profile, vm: vm)

                Divider().background(Color.mlmEdgeSubtle)

                // Progress OR Preview stats
                if let svc = syncService, svc.isRunning, svc.currentProfileId == profile.id {
                    SyncProgressSection(syncService: svc, onCancel: { vm?.cancelSync() })
                } else if let preview = vm?.preview {
                    previewStatsSection(preview)
                }

                // Result section (after sync completes)
                if let result = vm?.lastResult {
                    resultSection(result)
                }
            }
            .padding(16)
        }
        .background(Color.mlmBase)
        .sheet(isPresented: $showPlaylistPicker) {
            PlaylistPickerSheet(vm: vm)
        }
        .sheet(isPresented: $showTrackPicker) {
            TrackPickerSheet(vm: vm)
        }
        .onReceive(NotificationCenter.default.publisher(for: .syncProfileDidChange)) { notification in
            guard let profileId = notification.userInfo?["profileId"] as? Int64,
                  profileId == profile.id else { return }
            Task { await vm?.loadPreview(for: profile) }
        }
        .task(id: profile.id) {
            await vm?.loadPreview(for: profile)
        }
    }

    private var headerSection: some View {
        HStack(spacing: 8) {
            VStack(alignment: .leading, spacing: 4) {
                Text(profile.name).font(MLMFont.title2).foregroundColor(.mlmInk)
                Text(profile.outputFolder).font(MLMFont.muted).foregroundColor(.mlmInkMuted).lineLimit(1)
            }
            Spacer()
            Button("Refresh") {
                Task { await vm?.loadPreview(for: profile) }
            }
            Button("Sync Now") {
                Task { await vm?.executeSync(for: profile) }
            }
            .buttonStyle(.borderedProminent)
            .disabled(vm?.isSyncing ?? false || !(vm?.preview?.hasInsufficientSpace == false))
        }
    }

    private var contentHeader: some View {
        HStack(spacing: 8) {
            Text("Content").font(MLMFont.bodyBold).foregroundColor(.mlmInk)
            Spacer()
            Button("Add Playlists…") { showPlaylistPicker = true }
                .buttonStyle(.borderless).tint(.mlmAccent)
            Button("Add Tracks…") { showTrackPicker = true }
                .buttonStyle(.borderless).tint(.mlmAccent)
        }
        .padding(.top, 8)
    }

    @ViewBuilder
    private func previewStatsSection(_ preview: SyncPreview) -> some View {
        // Reuse the existing previewSection layout from SyncView (read SyncView.swift lines ~291-332)
        // Copy the stat-card HStack pattern here
        HStack(spacing: 24) {
            statCard(value: "\(preview.filesToAdd.count)", label: "To Add", color: .green)
            statCard(value: "\(preview.filesToRemove.count)", label: "To Remove", color: .mlmError)
            // Add size stat cards if available in SyncPreview
        }
    }

    private func statCard(value: String, label: String, color: Color) -> some View {
        VStack(spacing: 4) {
            Text(value)
                .font(.system(size: 24, weight: .semibold, design: .monospaced))
                .foregroundColor(color)
            Text(label).font(MLMFont.muted).foregroundColor(.mlmInkMuted)
        }
    }

    @ViewBuilder
    private func resultSection(_ result: SyncResult) -> some View {
        VStack(alignment: .leading, spacing: 8) {
            Divider().background(Color.mlmEdgeSubtle)
            HStack(spacing: 4) {
                Image(systemName: "checkmark.circle.fill").foregroundColor(.green)
                Text("\(result.syncedCount) synced")
                    .font(MLMFont.body).foregroundColor(.mlmInk)
                if result.failedCount > 0 {
                    Text(",").font(MLMFont.body).foregroundColor(.mlmInkMuted)
                    Image(systemName: "xmark.circle.fill").foregroundColor(.mlmError)
                    Text("\(result.failedCount) failed")
                        .font(MLMFont.body).foregroundColor(.mlmError)
                }
            }
            if result.failedCount > 0 {
                SyncFailedDisclosure(failedTracks: result.failedTracks, vm: vm)
            }
        }
    }
}
```

```swift
import SwiftUI

/// Collapsible settings form for a sync profile (D-02).
/// GREENFIELD — no existing Form+Toggle+Picker pattern in MLM (per PATTERNS.md).
struct SyncSettingsForm: View {
    let profile: SyncProfile
    let vm: SyncViewModel?
    @State private var isExpanded: Bool = false

    // Local mirrors for responsive toggle feedback before async commit
    @State private var localM3U8: Bool
    @State private var localTranscodeMode: String
    @State private var localFat32: Bool
    @State private var localCleanup: Bool

    init(profile: SyncProfile, vm: SyncViewModel?) {
        self.profile = profile
        self.vm = vm
        _localM3U8 = State(initialValue: profile.generateM3U8)
        _localTranscodeMode = State(initialValue: profile.transcodeMode)
        _localFat32 = State(initialValue: profile.fat32SafePaths)
        _localCleanup = State(initialValue: profile.cleanupRemovedFiles)
    }

    var body: some View {
        DisclosureGroup(isExpanded: $isExpanded) {
            Form {
                Toggle("Generate M3U8 Playlists", isOn: $localM3U8)
                    .onChange(of: localM3U8) { _, newValue in
                        Task { await vm?.updateProfileSettings(generateM3U8: newValue) }
                    }
                Picker("Transcode Mode", selection: $localTranscodeMode) {
                    Text("Keep Originals").tag("keep_originals")
                    Text("248 kbps AAC").tag("aac_248")
                    Text("320 kbps AAC").tag("aac_320")
                }
                .onChange(of: localTranscodeMode) { _, newValue in
                    Task { await vm?.updateProfileSettings(transcodeMode: newValue) }
                }
                Toggle("FAT32-Safe Filenames", isOn: $localFat32)
                    .onChange(of: localFat32) { _, newValue in
                        Task { await vm?.updateProfileSettings(fat32SafePaths: newValue) }
                    }
                Toggle("Delete Removed Files from Destination", isOn: $localCleanup)
                    .onChange(of: localCleanup) { _, newValue in
                        Task { await vm?.updateProfileSettings(cleanupRemovedFiles: newValue) }
                    }
            }
            .formStyle(.grouped)
            .padding(.vertical, 8)
        } label: {
            Text("Settings")
                .font(MLMFont.sectionLabel)
                .foregroundColor(.mlmInkMuted)
        }
        .padding(.horizontal, 16)
        .background(Color.mlmSurface)
        // Re-sync local state if parent profile changes (e.g., after remote update)
        .onChange(of: profile.id) { _, _ in
            localM3U8 = profile.generateM3U8
            localTranscodeMode = profile.transcodeMode
            localFat32 = profile.fat32SafePaths
            localCleanup = profile.cleanupRemovedFiles
        }
    }
}
```

```swift
import SwiftUI

/// Two DisclosureGroups: "Playlists (N)" and "Tracks (N)" with hover-trash removal.
/// Pattern: PlaylistDetailView.swift line 251-307 (trackRow + hover-trash + contextMenu).
struct SyncContentSections: View {
    let profile: SyncProfile
    let vm: SyncViewModel?
    @State private var playlistsExpanded: Bool = false
    @State private var tracksExpanded: Bool = false
    @State private var hoveredPlaylistId: Int64? = nil
    @State private var hoveredTrackId: Int64? = nil

    var body: some View {
        VStack(spacing: 0) {
            // Playlists section
            DisclosureGroup(isExpanded: $playlistsExpanded) {
                if let playlists = vm?.profilePlaylists {
                    ForEach(playlists) { playlist in
                        playlistRow(playlist)
                    }
                }
            } label: {
                HStack {
                    Text("Playlists (\(vm?.profilePlaylists?.count ?? 0))")
                        .font(MLMFont.bodyBold)
                        .foregroundColor(.mlmInk)
                    Spacer()
                }
                .padding(.vertical, 8)
            }
            .padding(.horizontal, 16)

            Divider().background(Color.mlmEdgeSubtle)

            // Tracks section
            DisclosureGroup(isExpanded: $tracksExpanded) {
                if let tracks = vm?.profileTracks {
                    ForEach(tracks) { track in
                        trackRow(track)
                    }
                }
            } label: {
                HStack {
                    Text("Tracks (\(vm?.profileTracks?.count ?? 0))")
                        .font(MLMFont.bodyBold)
                        .foregroundColor(.mlmInk)
                    Spacer()
                }
                .padding(.vertical, 8)
            }
            .padding(.horizontal, 16)
        }
    }

    private func playlistRow(_ playlist: Playlist) -> some View {
        HStack(spacing: 8) {
            VStack(alignment: .leading, spacing: 2) {
                Text(playlist.name)
                    .font(MLMFont.body)
                    .foregroundColor(.mlmInk)
                    .lineLimit(1)
                // Track count if available
            }
            Spacer()
            // Hover trash icon
            if hoveredPlaylistId == playlist.id {
                Button {
                    Task { await vm?.removePlaylists([playlist.id ?? -1]) }
                } label: {
                    Image(systemName: "trash")
                        .foregroundColor(.mlmError)
                        .font(.system(size: 12))
                }
                .buttonStyle(.borderless)
                .padding(.trailing, 8)
            }
        }
        .padding(.vertical, 6)
        .contentShape(Rectangle())
        .onHover { hovering in
            withAnimation(.easeInOut(duration: 0.15)) {
                hoveredPlaylistId = hovering ? playlist.id : nil
            }
        }
        .background(hoveredPlaylistId == playlist.id ? Color.mlmSurface : Color.clear)
        .contextMenu {
            Button(role: .destructive) {
                Task { await vm?.removePlaylists([playlist.id ?? -1]) }
            } label: {
                Label("Remove from Profile", systemImage: "minus.circle")
            }
        }
    }

    private func trackRow(_ track: Track) -> some View {
        HStack(spacing: 8) {
            VStack(alignment: .leading, spacing: 2) {
                Text("\(track.artist) — \(track.title)")
                    .font(MLMFont.body)
                    .foregroundColor(.mlmInk)
                    .lineLimit(1)
            }
            Spacer()
            if hoveredTrackId == track.id {
                Button {
                    Task { await vm?.removeTracks([track.id ?? -1]) }
                } label: {
                    Image(systemName: "trash")
                        .foregroundColor(.mlmError)
                        .font(.system(size: 12))
                }
                .buttonStyle(.borderless)

                Task { await vm?.removeTracks([track.id ?? -1]) }
```

```swift
import SwiftUI

/// Live progress section shown during sync (D-12).
/// Replaces preview stats in SyncProfileDetailView while syncService.isRunning.
/// Pattern: SyncView.previewSection (SyncView.swift:291-332) for layout shape.
struct SyncProgressSection: View {
    let syncService: SyncService
    let onCancel: () -> Void

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            // Counter + Cancel row
            HStack {
                Text("\(syncService.processed) of \(syncService.total)")
                    .font(MLMFont.bodyBold)
                    .foregroundColor(.mlmInk)
                Spacer()
                Button(role: .destructive) {
                    onCancel()
                } label: {
                    Label("Cancel", systemImage: "xmark.circle")
                }
                .buttonStyle(.borderless)
                .tint(.mlmAccent)
            }

            // Full-width linear progress bar
            ProgressView(value: syncService.progress)
                .progressViewStyle(.linear)
                .tint(.mlmAccent)

            // Current file being processed
            if !syncService.currentFile.isEmpty {
                Text(syncService.currentFile)
                    .font(MLMFont.muted)
                    .foregroundColor(.mlmInkSecondary)
                    .lineLimit(1)
                    .truncationMode(.middle)
            }
        }
        .padding(16)
        .background(Color.mlmSurface)
        .clipShape(RoundedRectangle(cornerRadius: 8))
    }
```
