import Testing
import GRDB
@testable import MLM

@Suite("PickerSheetTests")
@MainActor
struct PickerSheetTests {

    private func makeVM() throws -> (DatabaseQueue, SyncViewModel) {
        let db = try DatabaseManager.inMemory()
        let syncRepo = SyncRepository(database: db)
        let trackRepo = TrackRepository(database: db)
        let cache = TranscodeCache(database: db, libraryRoot: URL(fileURLWithPath: "/tmp"))
        let svc = SyncService(syncRepository: syncRepo, trackRepository: trackRepo, transcodeCache: cache)
        let vm = SyncViewModel(syncRepository: syncRepo, syncService: svc)
        return (db, vm)
    }

    @Test func addPlaylistsIsIdempotent() async throws {
        let (db, vm) = try makeVM()
        // Create a profile and a playlist
        try await db.write { db in
            try db.execute(sql: "INSERT INTO sync_profiles (name, output_folder, playlist_path_prefix) VALUES ('P', '/tmp', '')")
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Club Set 2026', 'regular')")
        }
        let profile = try await db.read { db in try SyncProfile.fetchOne(db)! }
        let playlistId: Int64 = try await db.read { db in
            (try Row.fetchOne(db, sql: "SELECT id FROM playlists"))?["id"] ?? -1
        }
        vm.selectedProfile = profile

        // Add same playlist twice
        await vm.addPlaylists([playlistId, playlistId])

        // Count rows in sync_profile_playlists — should be 1, not 2
        let count: Int = try await db.read { db in
            try Int.fetchOne(db, sql: "SELECT COUNT(*) FROM sync_profile_playlists WHERE profile_id = ?",
                             arguments: [profile.id!])!
        }
        #expect(count == 1)  // INSERT OR IGNORE — second insert is a no-op
        _ = db
    }

    @Test func searchFilterEmptyQueryReturnsAll() async throws {
        let (db, vm) = try makeVM()
        // Insert 3 playlists
        try await db.write { db in
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Club Set 2026', 'regular')")
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Focus', 'regular')")
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Workout', 'regular')")
        }
        let all = await vm.allAvailablePlaylists()
        #expect(all.count == 3)
        _ = db
    }

    @Test func searchFilterMatchesCaseInsensitive() async throws {
        let (db, vm) = try makeVM()
        try await db.write { db in
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Club Set 2026', 'regular')")
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('Focus', 'regular')")
        }
        let all = await vm.allAvailablePlaylists()
        // Filter locally (mirrors PlaylistPickerSheet logic)
        let query = "club"
        let filtered = all.filter { $0.name.localizedCaseInsensitiveContains(query) }
        #expect(filtered.count == 1)
        #expect(filtered.first?.name == "Club Set 2026")
        _ = db
    }

    @Test func addTracksPostsNotification() async throws {
        let (db, vm) = try makeVM()
        try await db.write { db in
            try db.execute(sql: "INSERT INTO sync_profiles (name, output_folder, playlist_path_prefix) VALUES ('P', '/tmp', '')")
        }
        let profile = try await db.read { db in try SyncProfile.fetchOne(db)! }
        vm.selectedProfile = profile

        var received = false
        let token = NotificationCenter.default.addObserver(forName: .syncProfileDidChange, object: nil, queue: nil) { _ in received = true }
        defer { NotificationCenter.default.removeObserver(token) }

        await vm.addTracks([42])  // trackId 42 doesn't exist; INSERT OR IGNORE
        #expect(received)
        _ = db
    }
}

import Testing

@Suite("SyncContextMenuTests")
@MainActor
struct SyncContextMenuTests {

    private func makeVM() throws -> (DatabaseQueue, SyncViewModel) {
        let db = try DatabaseManager.inMemory()
        let syncRepo = SyncRepository(database: db)
        let trackRepo = TrackRepository(database: db)
        let cache = TranscodeCache(database: db, libraryRoot: URL(fileURLWithPath: "/tmp"))
        let svc = SyncService(syncRepository: syncRepo, trackRepository: trackRepo, transcodeCache: cache)
        let vm = SyncViewModel(syncRepository: syncRepo, syncService: svc)
        return (db, vm)
    }

    @Test func profilesEmptyInitially() async throws {
        let (_, vm) = try makeVM()
        await vm.loadProfiles()
        #expect(vm.profiles.isEmpty)
    }

    @Test func profilesPopulatedAfterCreate() async throws {
        let (db, vm) = try makeVM()
        await vm.createProfile(name: "iPod Sync", outputFolder: "/tmp")
        await vm.loadProfiles()
        #expect(vm.profiles.count == 1)
        #expect(vm.profiles.first?.name == "iPod Sync")
        _ = db
    }

    @Test func addPlaylistToProfileViaContextMenuAction() async throws {
        let (db, vm) = try makeVM()
        // Create profile and playlist
        try await db.write { db in
            try db.execute(sql: "INSERT INTO sync_profiles (name, output_folder, playlist_path_prefix) VALUES ('P', '/tmp', '')")
            try db.execute(sql: "INSERT INTO playlists (name, category) VALUES ('MyPL', 'regular')")
        }
        let profile = try await db.read { db in try SyncProfile.fetchOne(db)! }
        let playlistId: Int64 = try await db.read { db in
            (try Row.fetchOne(db, sql: "SELECT id FROM playlists"))?["id"] ?? -1
        }
        vm.selectedProfile = profile

        // Simulate context menu "Sync to Profile" action — calls addPlaylists
        await vm.addPlaylists([playlistId])

        // Verify the playlist is now in the profile
        let count: Int = try await db.read { db in
            try Int.fetchOne(db, sql: "SELECT COUNT(*) FROM sync_profile_playlists WHERE profile_id = ?",
                             arguments: [profile.id!])!
        }
        #expect(count == 1)
        _ = db
    }

    @Test func syncProfilesHaveNewToggleColumns() async throws {
        let (db, vm) = try makeVM()
        await vm.createProfile(name: "Folder Sync", outputFolder: "/tmp")
        let profiles = try await db.read { db in try SyncProfile.fetchAll(db) }
        let p = profiles.first!

        // Default values per D-01
        #expect(p.generateM3U8 == false)
        #expect(p.transcodeMode == "keep_originals")
        #expect(p.fat32SafePaths == true)
        #expect(p.cleanupRemovedFiles == true)
        _ = db
    }
}
