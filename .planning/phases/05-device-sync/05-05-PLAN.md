```rust
use tauri::State;
use std::sync::Mutex;
use rusqlite::Connection;
use std::path::PathBuf;
use crate::sync::{self, SyncProfile, SyncPreview, SyncResult, RockboxDevice};
use crate::models::sync::{SyncProfileDto, FilterRuleDto};
use crate::database::connection::get_connection;

/// Create a new sync profile
#[tauri::command]
pub async fn create_sync_profile(
    name: String,
    output_folder: String,
) -> Result<i64, String> {
    let conn = get_connection().map_err(|e| e.to_string())?;

    tokio::task::spawn_blocking(move || {
        sync::profile::create_sync_profile(&conn, &name, PathBuf::from(&output_folder))
            .map_err(|e| e.to_string())
    })
    .await
    .map_err(|e| e.to_string())?
}

/// List all sync profiles
#[tauri::command]
pub async fn list_sync_profiles() -> Result<Vec<SyncProfileDto>, String> {
    let conn = get_connection().map_err(|e| e.to_string())?;

    tokio::task::spawn_blocking(move || {
        let profiles = sync::profile::list_sync_profiles(&conn)
            .map_err(|e| e.to_string())?;

        // Convert to DTOs with computed counts
        let dtos: Vec<_> = profiles.iter()
            .map(|p| SyncProfileDto::from_profile(&conn, p))
            .collect::<Result<Vec<_>, _>>()
            .map_err(|e| e.to_string())?;

        Ok(dtos)
    })
    .await
    .map_err(|e| e.to_string())?
}

/// Get a specific sync profile
#[tauri::command]
pub async fn get_sync_profile(profile_id: i64) -> Result<SyncProfileDto, String> {
    let conn = get_connection().map_err(|e| e.to_string())?;

    tokio::task::spawn_blocking(move || {
        let profile = sync::profile::get_sync_profile(&conn, profile_id)
            .map_err(|e| e.to_string())?;

        SyncProfileDto::from_profile(&conn, &profile)
            .map_err(|e| e.to_string())
    })
    .await
    .map_err(|e| e.to_string())?
}
```

```tsx
import React, { useState, useEffect } from 'react';
import { invoke } from '@tauri-apps/api/core';
import { Button } from './ui/button';
import { Input } from './ui/input';
import { Card, CardHeader, CardTitle, CardContent } from './ui/card';

interface SyncProfileDto {
  id: number;
  name: string;
  output_folder: string;
  track_count: number;
  manual_track_count: number;
  playlist_count: number;
  rule_count: number;
  date_created: string;
  date_modified: string;
}

export function SyncProfiles() {
  const [profiles, setProfiles] = useState<SyncProfileDto[]>([]);
  const [newProfileName, setNewProfileName] = useState('');
  const [newProfileFolder, setNewProfileFolder] = useState('');
  const [isCreating, setIsCreating] = useState(false);

  useEffect(() => {
    loadProfiles();
  }, []);

  const loadProfiles = async () => {
    try {
      const result = await invoke<SyncProfileDto[]>('list_sync_profiles');
      setProfiles(result);
    } catch (error) {
      console.error('Failed to load sync profiles:', error);
    }
  };

  const createProfile = async () => {
    if (!newProfileName.trim() || !newProfileFolder.trim()) {
      return;
    }

    setIsCreating(true);
    try {
      await invoke('create_sync_profile', {
        name: newProfileName,
        outputFolder: newProfileFolder,
      });

      setNewProfileName('');
      setNewProfileFolder('');
      await loadProfiles();
    } catch (error) {
      console.error('Failed to create profile:', error);
    } finally {
      setIsCreating(false);
    }
  };

  const deleteProfile = async (profileId: number) => {
    if (!confirm('Delete this sync profile?')) return;

    try {
      await invoke('delete_sync_profile', { profileId });
      await loadProfiles();
    } catch (error) {
      console.error('Failed to delete profile:', error);
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex justify-between items-center">
        <h2 className="text-2xl font-bold">Sync Profiles</h2>
      </div>

      {/* Create Profile Form */}
      <Card>
        <CardHeader>
          <CardTitle>Create New Profile</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="space-y-4">
            <Input
              placeholder="Profile Name (e.g., iPod, iPhone)"
              value={newProfileName}
              onChange={(e) => setNewProfileName(e.target.value)}
            />
            <Input
              placeholder="Output Folder Path"
              value={newProfileFolder}
              onChange={(e) => setNewProfileFolder(e.target.value)}
            />
            <Button onClick={createProfile} disabled={isCreating}>
              {isCreating ? 'Creating...' : 'Create Profile'}
            </Button>
          </div>
        </CardContent>
      </Card>

      {/* Profile List */}
      <div className="grid gap-4">
        {profiles.map((profile) => (
          <Card key={profile.id}>
            <CardHeader>
              <CardTitle className="flex justify-between items-center">
                <span>{profile.name}</span>
                <Button
                  variant="destructive"
                  size="sm"
                  onClick={() => deleteProfile(profile.id)}
                >
                  Delete
                </Button>
              </CardTitle>
            </CardHeader>
            <CardContent>
              <div className="space-y-2 text-sm">
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Output Folder:</span>
                  <span className="font-mono">{profile.output_folder}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Total Tracks:</span>
                  <span>{profile.track_count}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Content Sources:</span>
                  <span>
                    {profile.manual_track_count} manual, {profile.playlist_count} playlists, {profile.rule_count} rules
                  </span>
                </div>
                <div className="flex gap-2 mt-4">
                  <Button
                    variant="outline"
                    size="sm"
                    onClick={() => {
                      // Navigate to profile detail view (to be implemented)
                      console.log('Edit profile:', profile.id);
                    }}
                  >
                    Edit Content
                  </Button>
                  <Button
                    variant="default"
                    size="sm"
                    onClick={() => {
                      // Navigate to sync preview (to be implemented in SyncPreview component)
                      console.log('Sync profile:', profile.id);
                    }}
                  >
                    Sync
                  </Button>
                </div>
              </div>
            </CardContent>
          </Card>
        ))}
      </div>

      {profiles.length === 0 && (
        <div className="text-center text-muted-foreground py-8">
          No sync profiles yet. Create one to get started.
        </div>
      )}
    </div>
  );
}
```
